;“Vamos las interrupciones!” una vez cada 2 segundos y
;no termine nunca.


EOI EQU 20h 
IMR EQU 21h 
INT1 EQU 25h 

CONT EQU 10h
COMP EQU 11h

N_TIMER EQU 15

org 60
dw RUT_TIMER ;la direccion de la rutina que atiende la interrupción

org 1000h 
msj db "Vamos las interrupciones!"
fin_msj db ?

org 3000h
RUT_TIMER: push ax
           INT 7
           mov al, 0
           out CONT, al
           mov al, EOI
           out EOI, al
           pop ax
           iret
         
org 2000h
CLI

mov al, 1111_1101b ;habilito interrupciones TIMER
out IMR, al

mov al, N_TIMER
out INT1, al

mov al, 0
out CONT, al

mov al, 2
out COMP, al

mov bx, offset msj
mov al, offset fin_msj - offset msj

STI

loop: jmp loop

int 0
