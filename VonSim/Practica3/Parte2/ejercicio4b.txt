;B) Idem A pero ahora el programa termina luego de 
;mostrar 5 veces el mensaje (es decir, luego de que
;presione 5 veces la tecla F10).

EOI EQU 20h 
IMR EQU 21h 
INT0 EQU 24h 

N_F10 EQU 15

org 60
dw RUT_F10 ;la direccion de la rutina que atiende la interrupción

org 1000h 
msj db "Vamos las interrupciones!"
fin_msj db ?
cont db 5

org 3000h
RUT_F10: push ax
         INT 7
         dec cont
         mov al, EOI
         out EOI, al
         pop ax
         iret
         
org 2000h
CLI

mov al, 1111_1110b ;habilito interrupciones F10
out IMR, al

mov al, N_F10
out INT0, al

mov bx, offset msj
mov al, offset fin_msj - offset msj

STI

loop: cmp cont, 0
      jnz loop

fin: mov al, 1111_1111b ;deshabilito las interrpciones
     out IMR, al
     
int 0
end
