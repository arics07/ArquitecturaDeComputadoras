;esperar a que el bit de busy sea 0 usando consulta de estado. Cuando eso suceda, imprimir “Libre” y
;terminar el programa.

PA EQU 30h
CA EQU 32h

ORG 1000h
no db “Libre”

ORG 2000H
mov al, 0FDh
out CA, al

poll: in al, PA
and al, 0000_0001b
jnz poll

mov bx, offset no
mov al, 5
int 7

int 0
end
